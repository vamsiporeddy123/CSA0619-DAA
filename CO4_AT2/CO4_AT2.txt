import java.util.*;

public class SmartWasteRouting {

    static class Edge {
        int to;
        double distance, trafficTime;

        Edge(int to, double distance, double trafficTime) {
            this.to = to;
            this.distance = distance;
            this.trafficTime = trafficTime;
        }

        double cost() {
            return distance + 0.5 * trafficTime;
        }
    }

    static class Node implements Comparable<Node> {
        int vertex;
        double distance;

        Node(int vertex, double distance) {
            this.vertex = vertex;
            this.distance = distance;
        }

        public int compareTo(Node other) {
            return Double.compare(this.distance, other.distance);
        }
    }

    static class Stop {
        int vertex;
        double waste;

        Stop(int vertex, double waste) {
            this.vertex = vertex;
            this.waste = waste;
        }
    }

    static final double INF = Double.POSITIVE_INFINITY;

    static void addEdge(List<Edge>[] graph, int u, int v,
                        double distance, double trafficTime) {
        graph[u].add(new Edge(v, distance, trafficTime));
        graph[v].add(new Edge(u, distance, trafficTime));
    }

    static double[] dijkstra(List<Edge>[] graph, int source) {
        int n = graph.length;
        double[] dist = new double[n];
        Arrays.fill(dist, INF);
        dist[source] = 0;

        PriorityQueue<Node> pq = new PriorityQueue<>();
        pq.add(new Node(source, 0));

        while (!pq.isEmpty()) {
            Node cur = pq.poll();

            if (cur.distance > dist[cur.vertex])
                continue;

            for (Edge e : graph[cur.vertex]) {
                double newDist = cur.distance + e.cost();

                if (newDist < dist[e.to]) {
                    dist[e.to] = newDist;
                    pq.add(new Node(e.to, newDist));
                }
            }
        }

        return dist;
    }

    static void buildRoutes(List<Edge>[] graph, int depot,
                            List<Stop> stops, double capacity) {

        int current = depot;
        double remaining = capacity;
        double totalCost = 0;

        Set<Integer> visited = new HashSet<>();

        System.out.println("Depot -> " + depot);

        while (visited.size() < stops.size()) {

            Stop best = null;
            double bestCost = INF;

            double[] dist = dijkstra(graph, current);

            for (Stop stop : stops) {
                if (visited.contains(stop.vertex))
                    continue;

                if (stop.waste > remaining)
                    continue;

                if (dist[stop.vertex] < bestCost) {
                    bestCost = dist[stop.vertex];
                    best = stop;
                }
            }

            if (best == null) {
                totalCost += dist[depot];
                System.out.println(" -> Depot (unload)");

                current = depot;
                remaining = capacity;
                continue;
            }

            totalCost += dist[best.vertex];
            remaining -= best.waste;
            current = best.vertex;
            visited.add(best.vertex);

            System.out.println(" -> " + best.vertex +
                    " [waste=" + best.waste +
                    ", remaining=" + remaining + "]");
        }

        if (current != depot) {
            double[] back = dijkstra(graph, current);
            totalCost += back[depot];
            System.out.println(" -> Depot");
        }

        System.out.printf("Total route cost = %.2f%n", totalCost);
    }

    public static void main(String[] args) {

        int n = 7;
        List<Edge>[] graph = new ArrayList[n];

        for (int i = 0; i < n; i++)
            graph[i] = new ArrayList<>();

        // distance, traffic-adjusted travel time
        addEdge(graph, 0, 1, 4, 3);
        addEdge(graph, 0, 2, 2, 2);
        addEdge(graph, 1, 3, 5, 5);
        addEdge(graph, 2, 3, 1, 2);
        addEdge(graph, 2, 4, 7, 3);
        addEdge(graph, 3, 5, 3, 2);
        addEdge(graph, 4, 5, 2, 4);
        addEdge(graph, 5, 6, 4, 3);
        addEdge(graph, 1, 6, 10, 8);

        List<Stop> stops = Arrays.asList(
            new Stop(1, 3),
            new Stop(3, 4),
            new Stop(4, 2),
            new Stop(5, 3),
            new Stop(6, 2)
        );

        double vehicleCapacity = 7;

        buildRoutes(graph, 0, stops, vehicleCapacity);
    }
}
